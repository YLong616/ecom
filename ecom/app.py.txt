        pass

        self.core.set_rts(self.rts_var.get())

    def _apply_language_font(self):

        try:
            self.output.configure(font=self.out_font)

    def _clamp_output(self):

        try:
            self.output.configure(height=22)
        except tk.TclError:

    def _bind_events(self):
        """Ctrl+滚轮缩放输出/输入字体（独立字体，互不影响）；输入框 Enter 发送。"""
        def make_wheel(fontob, widget):
            def on_wheel(event):
                if event.state & 0x0004:  # Ctrl
                    delta = 1 if event.delta > 0 else -1
                    size = max(8, min(28, fontob.cget("size") + delta))
                    if size != fontob.cget("size"):
                        fontob.configure(size=size)
                        widget.configure(font=fontob)
                    return "break"
                return None
            return on_wheel

        def on_enter(event):
            self._send()
            return "break"

        self.output.bind("<Control-MouseWheel>", make_wheel(self.out_font, self.output))
        self.output.bind("<Control-Button-4>", lambda e, f=self.out_font, w=self.output: self._zoom_font(f, w, 1))
        self.output.bind("<Control-Button-5>", lambda e, f=self.out_font, w=self.output: self._zoom_font(f, w, -1))
        self.input.bind("<Control-MouseWheel>", make_wheel(self.in_font, self.input))
        self.input.bind("<Control-Button-4>", lambda e, f=self.in_font, w=self.input: self._zoom_font(f, w, 1))
        self.input.bind("<Control-Button-5>", lambda e, f=self.in_font, w=self.input: self._zoom_font(f, w, -1))
        self.input.bind("<Return>", on_enter)
        self.input.bind("<KP_Enter>", on_enter)
        self.input.bind("<Control-Right>", self._move_word_right)
        self.input.bind("<Control-Left>", self._move_word_left)

    def _zoom_font(self, fontob, widget, delta):
        size = max(8, min(28, fontob.cget("size") + delta))
        if size != fontob.cget("size"):
            fontob.configure(size=size)
            widget.configure(font=fontob)

    def _move_word(self, widget, direction):
        """Ctrl+方向键：按「单词/分隔符」分段移动光标，而非直接跳到首/尾。"""
        try:
            row, col = widget.index("insert").split(".")
            row = int(row)
            col = int(col)
            line = widget.get(f"{row}.0", f"{row}.end")
            clen = len(line)

            def cls(ch):
                return "w" if (ch.isalnum() or ch == "_") else "o"

            if direction > 0:
                if col >= clen:
                    widget.mark_set("insert", f"{row + 1}.0")
                    return
                cur = cls(line[col])
                ncol = col
                while ncol < clen and cls(line[ncol]) == cur:
                    ncol += 1
                widget.mark_set("insert", f"{row}.{ncol}")
            else:
                if col <= 0:
                    if row > 1:
                        prev = int(widget.index(f"{row - 1}.end").split(".")[1])
                        widget.mark_set("insert", f"{row - 1}.{prev}")
                    return
                cur = cls(line[col - 1])
                ncol = col
                while ncol > 0 and cls(line[ncol - 1]) == cur:
                    ncol -= 1
                widget.mark_set("insert", f"{row}.{ncol}")
        except tk.TclError:
            pass

    def _move_word_right(self, event):
        self._move_word(self.input, 1)
        return "break"

    def _move_word_left(self, event):
        self._move_word(self.input, -1)
        return "break"

    def _build_menu(self):
        # 下拉菜单本体（点击「设置」标题按钮时弹出）
        self.settings_menu = tk.Menu(self.root, tearoff=0, font=self.ui_font)
        self.lang_menu = tk.Menu(self.settings_menu, tearoff=0, font=self.ui_font)
        self._lang_var = tk.StringVar(value=self.lang)
        for code in i18n.SUPPORTED_LANGUAGES:
            self.lang_menu.add_radiobutton(
                label=i18n.lang_label(code), value=code, variable=self._lang_var,
                command=lambda c=code: self._set_language(c))
        self.settings_menu.add_cascade(label=self._t("menu_language"), menu=self.lang_menu)
        self.settings_menu.add_separator()
        self.settings_menu.add_command(label=self._t("menu_about"), command=self._open_about)
        self.settings_menu.add_separator()
        self.settings_menu.add_command(label=self._t("menu_exit"), command=self._on_close)

        # 模式切换菜单（AT / 压力）
        self.mode_menu = tk.Menu(self.root, tearoff=0, font=self.ui_font)
        # 帮助菜单
        self.help_menu = tk.Menu(self.root, tearoff=0, font=self.ui_font)
        self.help_menu.add_command(label=self._t("help_guide"), command=self._open_help_guide)
        for code, lbl in (("at", "mode_at"), ("stress", "mode_stress")):
            self.mode_menu.add_radiobutton(label=self._t(lbl), value=code,
                                           variable=self._mode_var,
                                           command=lambda c=code: self._switch_mode(c))

        # 顶部标题栏：左侧「设置」大字标题按钮；左间距与「串口配置」标题对齐；下方留出间距
        if getattr(self, "_header_built", None) is None:
            self.header = ttk.Frame(self.root, style="TFrame")
            self.header.pack(side="top", fill="x", padx=0, pady=(0, 6))
            self.settings_btn = ttk.Label(self.header, text=self._t("menu_settings"),
                                          font=self.menu_font, cursor="hand2")
            self.settings_btn.pack(side="left", padx=(10, 6), pady=(8, 6))
            self.settings_btn.bind("<Button-1>", self._popup_settings)
            self._register("menu_settings", self.settings_btn)
            self.mode_btn = ttk.Label(self.header, text=self._t("mode_switch"),
                                      font=self.menu_font, cursor="hand2")
            self.mode_btn.pack(side="left", padx=(0, 6), pady=(8, 6))
            self.mode_btn.bind("<Button-1>", self._popup_mode)
            self._register("mode_switch", self.mode_btn)
            # 宏定义入口（放在「帮助」左边；独立窗口，不动主界面布局）
            self.macro_btn = ttk.Label(self.header, text=self._t("menu_macro"),
                                       font=self.menu_font, cursor="hand2")
            self.macro_btn.pack(side="left", padx=(0, 6), pady=(8, 6))
            self.macro_btn.bind("<Button-1>", lambda e: self._open_macros())
            self._register("menu_macro", self.macro_btn)
            self.help_btn = ttk.Label(self.header, text=self._t("menu_help"),
                                      font=self.menu_font, cursor="hand2")
            self.help_btn.pack(side="left", padx=(0, 6), pady=(8, 6))
            self.help_btn.bind("<Button-1>", self._popup_help)
            self._register("menu_help", self.help_btn)

    # ---------------- 宏定义（参数自定义） ----------------
    def macro_items(self):
        items = self.config.data.get("macros")
        return items if isinstance(items, list) else []

    def save_macro_items(self, rows):
        """整表保存（只有点保存才落盘；未保存关闭即作废）。"""
        self.config.data["macros"] = [{"key": str(r.get("key", "")),
                                       "value": str(r.get("value", ""))}
                                      for r in rows]
        self.config.save()

    def macro_map(self):
        return macro.parse_macros(self.macro_items())

    def expand_macros(self, text):
        """把 {name} 展开成实际内容。返回 (文本, 错误列表)；有错误时不要发送。"""
        return macro.expand(text, self.macro_map())

    def report_macro_errors(self, errors):
        for key, err in errors:
            self._append_input_msg(self._t("macro_expand_err", name=key, e=err) + "\n")

    def _open_macros(self):
        from .macro_panel import show_macros
        return show_macros(self)

    def _popup_help(self, event=None):
        try:
            self.help_menu.tk_popup(event.x_root, event.y_root)
        finally:
            self.help_menu.grab_release()

    def _open_help_guide(self):
        try:
            from .help_viewer import show_help
            show_help(self.root, self.lang, ui_font=self.ui_font,
                      menu_font=self.menu_font, out_font=self.out_font)
        except Exception as exc:
            self._append_input_msg(self._t("help_guide") + ": " + str(exc) + "\n")

    def _open_about(self):
        win = tk.Toplevel(self.root)
        win.title(self._t("menu_about"))
        win.configure(bg=P["bg"])
        w, h = 420, 268
        win.geometry(f"{w}x{h}")
        win.resizable(False, False)
        try:
            win.transient(self.root)
        except tk.TclError:
            pass
        win.update_idletasks()
        x = self.root.winfo_rootx() + (self.root.winfo_width() - w) // 2
        y = self.root.winfo_rooty() + (self.root.winfo_height() - h) // 2
        win.geometry(f"{w}x{h}+{max(x, 0)}+{max(y, 0)}")
        card = ttk.Frame(win, style="Surface.TFrame")
        card.pack(fill="both", expand=True, padx=16, pady=14)
        ttk.Label(card, text=self._t("app_title"), font=self.menu_font,
                  style="TLabel").pack(pady=(0, 2))
        ttk.Label(card, text="eCOM " + __version__, font=self.ui_font,
                  style="Muted.TLabel").pack(pady=(0, 8))
        ttk.Label(card, text=self._t("about_desc"), font=self.ui_font,
                  justify="center", style="Muted.TLabel").pack(pady=(0, 10))
        ttk.Label(card, text=self._t("about_author"), font=self.ui_font,
                  style="Muted.TLabel").pack(pady=(0, 12))
        row = ttk.Frame(card, style="Surface.TFrame")
        row.pack()
        ttk.Button(row, text=self._t("about_open_guide"), style="Primary.TButton",
                   command=self._open_help_guide).pack(side="left", padx=(0, 8))
        ttk.Button(row, text=self._t("help_close"),
                   command=win.destroy).pack(side="left")

    def _get_stress_view(self):
        if getattr(self, "stress_view", None) is None:
            from .stress_panel import StressPanel
            self.stress_view = StressPanel(self.content, self)
        return self.stress_view

    def _switch_mode(self, mode):
        if mode == self._mode:
            return
        if mode == "stress":
            self.at_view.pack_forget()
            sv = self._get_stress_view()
            sv.pack(fill="both", expand=True)
        else:
            if getattr(self, "stress_view", None) is not None:
                try:
                    self.stress_view._stop_run()
                except Exception:
                    pass
                self.stress_view.pack_forget()
            self.at_view.pack(fill="both", expand=True)
        self._mode = mode
        self._mode_var.set(mode)

    def _build_widgets(self):
        self._build_body()
        self._build_statusbar()
        self._sync_page_scroll()      # 首帧先定一次尺寸（此时尚未收到 <Configure>）

    def _label(self, parent, key, **kw):
        w = ttk.Label(parent, text=self._t(key), **kw)
        self._register(key, w)
        return w

    def _button(self, parent, key, command, style=None, **kw):
        w = ttk.Button(parent, text=self._t(key), command=command)
        if style:
            w.configure(style=style)
        self._register(key, w)
        if kw:
            w.pack(**kw)
        else:
            w.pack()
        return w

    # ---------------- 主页滚动（窗口缩小时可滚着看全部内容） ----------------
    def _sync_page_scroll(self):
        """内容不超出可视区时铺满窗口并藏起滚动条；超出时才显示并可滚。"""
        if self._page_syncing:
            return
        self._page_syncing = True
        try:
            cv, win = self.page_canvas, self.content
            need_h = win.winfo_reqheight()
            vw, vh = cv.winfo_width(), cv.winfo_height()
            # 宽：以历史最小宽 PAGE_MIN_W 为下限（默认窗口下内容仍铺满 -> 不出现横向滚动条）；
            #     只有比它更窄才横向滚动。若用「自然宽」当下限，默认尺寸下就会多出一条横向滚动条。
            # 高：以内容自然高为下限 -> 窗口变矮时内容不被裁切，直接纵向滚动。
            full_w = max(vw, PAGE_MIN_W)
            full_h = max(vh, need_h)
            cv.itemconfigure(self._page_win, width=full_w, height=full_h)
            cv.configure(scrollregion=(0, 0, full_w, full_h))
            if need_h > vh:
                self.page_vbar.grid()
            else:
                self.page_vbar.grid_remove()
                cv.yview_moveto(0)
            if full_w > vw:
                self.page_hbar.grid()
            else:
                self.page_hbar.grid_remove()
                cv.xview_moveto(0)
        finally:
            self._page_syncing = False

    def _page_wheel_target(self, event):
        """判断这次滚轮是否该翻主页面（对话框、自带滚动的控件不抢）。"""
        w = getattr(event, "widget", None)
        if w is None:
            return False
        try:
            if w.winfo_toplevel() is not self.root:
                return False          # 弹窗里的滚轮不动主页面
        except tk.TclError:
            return False
        if w is self.page_canvas:
            return True
        return w.winfo_class() not in ("Text", "Canvas", "Listbox", "Treeview",
                                       "TCombobox", "TSpinbox", "Spinbox",
                                       "Scrollbar", "TScrollbar")

    def _on_page_wheel(self, event):
        if not self._page_wheel_target(event):
            return None
        self.page_canvas.yview_scroll(-3 if event.delta > 0 else 3, "units")
        return "break"

    def _on_page_wheel_h(self, event):
        """Shift+滚轮：横向滚动（看右栏）。"""
        if not self._page_wheel_target(event):
            return None
        self.page_canvas.xview_scroll(-3 if event.delta > 0 else 3, "units")
        return "break"

    def _mm_holds_port(self, port):
        """POSIX 下检测 /dev/ 串口是否被 ModemManager 占用；非 POSIX 恒 False。"""
        if os.name != "posix":
            return False
        if not (port or "").startswith("/dev/"):
            return False
        try:
            p = subprocess.run(["fuser", "-v", port],
                               capture_output=True, text=True, timeout=3)
            out = (p.stdout or "") + (p.stderr or "")
            return "ModemManager" in out
        except Exception:
            return False

    # ---------- 串口操作 ----------
    def _refresh_ports(self, flash=True):
        self._port_map = {}
        displays = []
        for device, display in self.core.list_ports():
            self._port_map[display] = device
            displays.append(display)
        self.port_combo["values"] = displays
        cur = self.port_var.get()
        if cur not in displays:
            matched = next((d for d, dev in self._port_map.items()
                            if dev == cur or d == cur), None)
            self.port_var.set(matched if matched else "")
        # 刷新成功：端口框未选中时自动选中第一个，并在状态栏提示
        if displays and not self.port_var.get():
            self.port_var.set(displays[0])
        if displays and flash:
            self._flash_status(self._t("refresh_ok", n=len(displays)))

    def _gather_params(self):
        baud = int(self.baud_var.get())
        data_bits = int(self.databits_var.get())
        parity = PARITY_MAP[self.parity_var.get()]
        stop_bits = float(self.stopbits_var.get())
        rtscts, xonxoff = FLOW_MAP[self.flow_var.get()]
        return baud, data_bits, parity, stop_bits, rtscts, xonxoff

    def _toggle_port(self):
        if self.core.is_open:
            self.core.close()
            self._active_port = None
            self._status_error = None
            self._update_status()
            return
        display = self.port_var.get().strip()
        if not display:
            self._append_input_msg(self._t("no_port") + "\n")
            return
        port = self._port_map.get(display, display)
        try:
            baud, data_bits, parity, stop_bits, rtscts, xonxoff = self._gather_params()
            self.core.open(port, baudrate=baud, bytesize=data_bits, parity=parity,
                           stopbits=stop_bits, rtscts=rtscts, xonxoff=xonxoff)
            self._active_port = port
            self._disconnecting = False
            self._presence_tick = 0
            self._apply_dtr_rts()
            if self._mm_holds_port(port):
                self._append_input_msg(self._t("help_mm_hint") + "\n")
            self._status_error = None
            self._update_status()
        except Exception as exc:
            self._append_input_msg(self._t("open_fail", e=exc) + "\n")
            self._set_status_error(str(exc))
            if self._mm_holds_port(port):
                self._append_input_msg(self._t("help_mm_hint") + "\n")

    def _send(self):
        text = self.input.get("1.0", "end-1c")
        self.send_text(text)

    def send_text(self, text):
        # 宏展开：未知宏/求值失败 -> 报错并阻止发送
        text, _errs = self.expand_macros(text)
        if _errs:
            self.report_macro_errors(_errs)
            return 0
        data_type = self.send_type_var.get()
        try:
            raw = codec.encode(text, data_type, self.send_suffix_var.get())
        except ValueError as exc:
            self._append_input_msg(self._t("encode_bad", e=exc) + "\n")
            return 0
        return self.send_bytes(raw)

    def send_bytes(self, raw):
        if not raw:
            return 0
        if not self.core.is_open:
            self._append_input_msg(self._t("no_port") + "\n")
            return 0
        try:
            n = self.core.write(raw)
            self._append_line("TX", raw)
            self._rx_last = time.monotonic()
            return n
        except Exception as exc:
            self._append_input_msg(self._t("send_fail", e=exc) + "\n")
            return 0

    # ---------- 接收轮询 ----------
    def _poll_rx(self):
        import time as _t
        try:
            got = False
            for kind, payload in self.core.drain():
                if kind == SerialCore.RX_DATA:
                    self._rx_buf.extend(payload)
                    got = True
                elif kind == SerialCore.RX_ERROR:
                    self._on_port_lost(str(payload))
                elif kind == SerialCore.RX_CLOSED:
                    self._on_port_lost()
            if got:
                self._rx_last = _t.monotonic()
                self._flush_rx_lines()          # 读一行即输出一行（完整行立即渲染）
            # 静默超时 → 补渲染残留的不完整尾部
            if self._rx_buf and (_t.monotonic() - self._rx_last) * 1000 >= self._rx_idle_ms:
                self._flush_rx()
            # 周期性探测：串口被拔出/设备消失时主动断开，恢复为“打开串口”
            if self.core.is_open and self._active_port:
                self._presence_tick += 1
                if self._presence_tick >= self._presence_check_interval:
                    self._presence_tick = 0
                    self._check_port_presence()
        finally:
            self._update_counters()
            self.root.after(self._poll_interval_ms, self._poll_rx)

    def _append_interp(self, text):
        """在输出区追加纯文本（禁用 insert 状态，辅助函数内部处理）。"""
        try:
            self.output.insert("end", text)
        except tk.TclError:
            return

    def _render_frame(self, direction, raw):
        """渲染一条完整消息（TX 或聚合后的 RX），统一黑色。"""
        now = time.time()
        dt = datetime.datetime.now()
        ts = "[%s:%03d]" % (dt.strftime("%Y-%m-%d_%H:%M:%S"), dt.microsecond // 1000)
        parts = []
        if self.show_ts.get():
            parts.append(ts)
        if self.show_dir.get():
            parts.append(direction)
        body = codec.display_body(raw, self.show_hex.get(), self.show_crlf.get())
        if body:
            parts.append(body)
        line = "  ".join(parts) + "\n"
        try:
            self.output.insert("end", line)
            self.output.see("end")
        except tk.TclError:
            return
        # Log 与输出框逐字节一致：把渲染进输出框的那一行原样写入（去掉末尾换行）
        if self.logger.active and direction in ("TX", "RX"):
            self.logger.log(line.rstrip("\n"))

    def _flush_rx_lines(self):
        """立即渲染已完成的整行（以 LF 结尾），不完整的尾部保留缓冲续接下一行。"""
        if not self._rx_buf:
            return
        data = bytes(self._rx_buf)
        self._rx_buf.clear()
        while True:
            idx = data.find(b'\n')
            if idx < 0:
                if data:
                    self._rx_buf.extend(data)
                break
            self._render_frame('RX', data[:idx + 1])
            data = data[idx + 1:]

    def _flush_rx(self):
        """若接收缓冲非空，按换行拆分为多条 RX（每行独立时间戳）并清空。"""
        if self._rx_buf:
            data = bytes(self._rx_buf)
            self._rx_buf.clear()
            while True:
                idx = data.find(b"\n")
                if idx < 0:
                    if data:
                        self._render_frame("RX", data)
                    break
                self._render_frame("RX", data[:idx + 1])
                data = data[idx + 1:]

    def _append_line(self, direction, raw):
        """TX 入口：先冲刷未渲染的 RX 缓冲，避免 TX 插到 RX 中间。"""
        if direction == "TX":
            self._flush_rx()
        self._render_frame(direction, raw)

    def _append_input_msg(self, text):
        try:
            self.input.configure(state="normal")
            self.input.insert("end", text)
            self.input.see("end")
        finally:
            self.input.configure(state="normal")

    # ---------- 状态 ----------
    def _set_dot(self, color):
        self.dot.delete("all")
        self.dot.create_oval(2, 2, 11, 11, fill=color, outline=color)

    def _set_status_error(self, msg):
        self._status_error = msg
        self._set_dot(P["danger"])
        self.status_label.config(text=self._t("status_error", e=msg))
        self.open_btn.configure(text=self._t("open"))

    def _update_status(self):
        if self.core.is_open:
            self._set_dot(P["success"])
            self.status_label.config(text=self._t("status_connected"))
            device = self._port_map.get(self.port_var.get(), self.port_var.get())
            summary = (f"{device} · {self.baud_var.get()}-{self.databits_var.get()}-"
                       f"{PARITY_MAP[self.parity_var.get()]}-{self.stopbits_var.get()}")
            self.connect_summary.config(text=summary)
            self.open_btn.configure(text=self._t("close"))
        else:
            self._set_dot(P["danger"] if self._status_error else P["idle"])
            self.status_label.config(
                text=self._t("status_error", e=self._status_error) if self._status_error
                else self._t("status_disconnected"))
            self.connect_summary.config(text="--")
            self.open_btn.configure(text=self._t("open"))

    def _check_port_presence(self):
        """打开期间定期探测当前端口是否仍在系统串口列表中。"""
        port = self._active_port
        if not port:
            return
        # loop:// 等虚拟端口不被 list_ports 枚举，跳过探测，避免误判断开
        if "://" in port:
            return
        present = any(dev == port for dev, _ in self.core.list_ports())
        if not present:
            self._on_port_lost(self._t("port_lost"))

    def _on_port_lost(self, reason=None):
        """设备被移除/读写失效 -> 主动断开连接，恢复为“打开串口”。"""
        if self._disconnecting:
            return
        self._disconnecting = True
        try:
            self.core.close()
            self._active_port = None
            self._flush_rx()
            msg = reason or self._t("port_lost")
            self._append_input_msg(msg + "\n")
            self._set_status_error(msg)
            self._refresh_ports(flash=False)   # 下拉框同步移除已失效端口
            self._update_status()
        finally:
            self._disconnecting = False

    def _flash_status(self, msg, ms=2200):
        """状态栏临时显示一条消息，几秒后恢复为真实连接状态。"""
        try:
            if getattr(self, "_flash_job", None):
                self.root.after_cancel(self._flash_job)
        except tk.TclError:
            pass
        self.status_label.config(text=msg)
        self._flash_job = self.root.after(ms, self._update_status)

    def _update_counters(self):
        tx, rx = self.core.tx_count, self.core.rx_count
        self.monitor_tx.config(text=self._t("tx", n=tx) + " " + self._t("byte"))
        self.monitor_rx.config(text=self._t("rx", n=rx) + " " + self._t("byte"))
        self.status_tx.config(text=f"TX {tx}")
        self.status_rx.config(text=f"RX {rx}")

    # ---------- 计数 / 清空 ----------
    def _clear_counters(self):
        self.core.reset_counters()
        self._update_counters()

    # ---------- 文本可复制（右键菜单 / Ctrl+A / Ctrl+C；只读但可选中） ----------
    def _install_select_menu(self, widget):
        menu = tk.Menu(widget, tearoff=0)
        menu.add_command(label=self._t("copy"), command=lambda: self._copy_selection(widget))
        menu.add_command(label=self._t("select_all"), command=lambda: self._select_all(widget))
        menu.add_command(label=self._t("copy_all"), command=lambda: self._copy_all(widget))
        widget._ctx_menu = menu  # 持有引用，避免被 GC
        widget.bind("<Button-3>", lambda e: menu.tk_popup(e.x_root, e.y_root))
        widget.bind("<Control-a>", self._on_select_all_key)
        # 复制/粘贴一律由本控件自己接管并 return "break"：
        # 一旦让 Tk 的类绑定（tk_textCopy / tk_textPaste）插手，剪贴板就转成
        # 「延迟渲染」——主线程忙于收 Log 时，外部程序粘贴会拿到空内容
        # （见 ecom/clipboard.py 的实测记录）。
        widget.bind("<Control-c>", self._on_copy_key)
        widget.bind("<<Copy>>", self._on_copy_key)
        widget.bind("<Control-v>", self._on_paste_key)
        widget.bind("<<Paste>>", self._on_paste_key)
        widget.bind("<<Cut>>", self._on_paste_key)
        # 输出框只读但可选中：放行 Ctrl 组合（复制/全选），拦截一切键入
        widget.bind("<KeyPress>", self._readonly_key)

    def _readonly_key(self, event):
        """输出框只读策略：Ctrl 组合键（复制/全选）放行，其余键入一律拦截。"""
        if event.state & 0x0004:  # ControlMask
            return None
        return "break"

    def _on_copy_key(self, event):
        """Ctrl+C / <<Copy>>：自己复制并 break，阻止 Tk 类绑定接管剪贴板。"""
        self._copy_selection(event.widget)
        return "break"

    def _on_paste_key(self, event):
        """只读框：Ctrl+V / <<Paste>> / <<Cut>> 一律不执行，避免改动控件内容。"""
        return "break"

    def _on_select_all_key(self, event):
        self._select_all(event.widget)
        return "break"

    def _copy_selection(self, widget):
        try:
            text = widget.get("sel.first", "sel.last")
        except tk.TclError:
            text = ""
        if text:
            self._copy_to_clipboard(widget, text)

    def _copy_to_clipboard(self, widget, text, done=None):
        """写入系统剪贴板（实现见 ecom/clipboard.py）。

        Windows：交给无 Tk 的独立进程 + Win32 立即渲染；**后台线程异步执行**，
        主线程立即返回 —— 复制长文本时若同步等待，紧接着的右键菜单/界面交互会卡顿。
        非 Windows（或打包版）在主线程同步回退 Tk 管理的剪贴板。
        **成功不写任何输入框提示**（只有失败才提示）；done(ok) 在主线程回调。
        """
        if len(text) > 3_000_000:
            self._append_input_msg(self._t("copy_big") + "\n")
        if os.name != "nt" or getattr(sys, "frozen", False):
            ok = set_clipboard(text, widget)
            if not ok:
                self._append_input_msg(self._t("copy_fail") + "\n")
            if done:
                done(ok)
            return ok
        self._ensure_clip_worker()
        self._clip_pending += 1
        self._clip_jobs.put((text, done))
        if self._clip_pending == 1:
            self.root.after(60, self._drain_clip_results)
        return True

    def _ensure_clip_worker(self):
        """按需启动唯一的剪贴板后台线程（线程里只调进程外写入，绝不碰 Tk）。"""
        if getattr(self, "_clip_jobs", None) is None:
            self._clip_jobs = queue.Queue()
            self._clip_results = queue.Queue()
            self._clip_pending = 0
            threading.Thread(target=self._clip_worker, daemon=True).start()

    def _clip_worker(self):
        while True:
            text, done = self._clip_jobs.get()
            try:
                ok = set_clipboard(text, None)
            except Exception:
                ok = False
            self._clip_results.put((ok, done))

    def _drain_clip_results(self):
        """主线程回收后台复制结果：成功静默，失败才提示。"""
        while True:
            try:
                ok, done = self._clip_results.get_nowait()
            except queue.Empty:
                break
            self._clip_pending -= 1
            if not ok:
                self._append_input_msg(self._t("copy_fail") + "\n")
            if done:
                try:
                    done(ok)
                except Exception:
                    pass
        if self._clip_pending > 0:
            self.root.after(60, self._drain_clip_results)

    def _select_all(self, widget):
        try:
            widget.tag_add("sel", "1.0", "end")
        except tk.TclError:
            pass

    def _copy_all(self, widget):
        try:
            widget.focus_set()          # 夺回焦点（info：避免按钮/滚动条抢占）
            text = widget.get("1.0", "end-1c")
        except tk.TclError:
            text = ""
        if text:
            n = len(text)
            self._copy_to_clipboard(widget, text,
                                    done=lambda ok: self._copy_all_done(ok, n))

    def _copy_all_done(self, ok, n):
        """复制全部完成（主线程）：成功只弹提示框确认，**不再往输入框写「已复制 N 字符」**。"""
        if not ok:
            return
        try:
            tk.messagebox.showinfo(self._t("copy"), self._t("copy_ok", n=n))
        except Exception:
            pass

    def _clear_output(self):
        self.output.delete("1.0", "end")

    def _clear_input(self):
        self.input.delete("1.0", "end")

    # ---------- Log ----------
    def _choose_log(self):
        path = filedialog.asksaveasfilename(
            title=self._t("log_select"), defaultextension=".log",
            initialfile=f"ecom_{datetime.datetime.now():%Y%m%d_%H%M%S}.log")
        if path:
            self._log_path = path
            self.log_path_var.set(path)
            self.config.set("log", "path", path)

    def _log_start(self):
        if not self._log_path:
            self._choose_log()
        if not self._log_path:
            return
        if self.logger.start(self._log_path):
            # 输出框已有的 Log 也一并写入（保存即时生效，并持续记录）
            existing = self.output.get("1.0", "end-1c")
            if existing and existing.strip():
                for ln in existing.split("\n"):
                    self.logger.log(ln)
            self._sync_log_state()
        else:
            self._append_input_msg(self._t("log_off") + "\n")

    def _log_toggle(self):
        if self.logger.active:
            self._log_stop()
        else:
            self._log_start()

    def _log_stop(self):
        if self.logger.active:
            self.logger.stop()
            self._sync_log_state()

    # ---------- 关闭 ----------
    def _bind_close(self):
        self.root.protocol("WM_DELETE_WINDOW", self._on_close)

    def _save_config(self):
        cfg = self.config.data
        cfg["serial"].update({
            "port": self.port_var.get(), "baudrate": self.baud_var.get(),
            "databits": self.databits_var.get(), "stopbits": self.stopbits_var.get(),
            "parity": self.parity_var.get(), "flow": self.flow_var.get()})
        cfg["display"].update({
            "timestamp": bool(self.show_ts.get()), "direction": bool(self.show_dir.get()),
            "crlf": bool(self.show_crlf.get()), "hex": bool(self.show_hex.get())})
        cfg["send"].update({
            "type": self.send_type_var.get(), "suffix": self.send_suffix_var.get()})
        cfg["language"] = self.lang
        try:
            self.config.save()
        except OSError:
            pass

    def _on_close(self):
        try:
            self.logger.stop()
        finally:
            self._save_config()
            self.core.close()
            self.root.destroy()


def run():
    root = tk.Tk()
    EcomApp(root, language="zh")
    root.mainloop()
