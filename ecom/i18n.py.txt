def get(lang, key, **fmt):
    try:
        text = I18N[lang][key]
    except KeyError:
        text = key
    if fmt:
        try:
            return text.format(**fmt)
        except (IndexError, KeyError, ValueError):
            # 占位符缺失，或文案里的花括号没转义 -> 退回原文，绝不因文案问题抛异常
            return text
    return text


def lang_label(lang):
    """语言的本名（不随当前语言翻译）。"""
    return "中文" if lang == "zh" else "English"
