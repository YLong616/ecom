# -*- coding: utf-8 -*-
"""用 PrintWindow + GetDIBits 把窗口抓成 24 位 BMP（屏幕 BitBlt 在本会话不可用时使用）。

grab_window(hwnd, path, rows=None) -> (宽, 高_已抓行数)
"""
import ctypes
import ctypes.wintypes as wt
import struct
import sys


class BITMAPINFOHEADER(ctypes.Structure):
    _fields_ = [("biSize", wt.DWORD), ("biWidth", ctypes.c_long),
                ("biHeight", ctypes.c_long), ("biPlanes", wt.WORD),
                ("biBitCount", wt.WORD), ("biCompression", wt.DWORD),
                ("biSizeImage", wt.DWORD), ("biXPelsPerMeter", ctypes.c_long),
                ("biYPelsPerMeter", ctypes.c_long), ("biClrUsed", wt.DWORD),
                ("biClrImportant", wt.DWORD)]


def grab_window(hwnd, path, rows=None, flags=2):
    u = ctypes.windll.user32
    g = ctypes.windll.gdi32
    rect = wt.RECT()
    u.GetWindowRect(hwnd, ctypes.byref(rect))
    w = rect.right - rect.left
    h = rect.bottom - rect.top
    if w <= 0 or h <= 0:
        raise RuntimeError("窗口尺寸异常: %sx%s" % (w, h))

    hdc = u.GetWindowDC(hwnd)
    memdc = g.CreateCompatibleDC(hdc)
    bmp = g.CreateCompatibleBitmap(hdc, w, h)
    g.SelectObject(memdc, bmp)
    ok = u.PrintWindow(hwnd, memdc, flags)
    if not ok:
        ok = u.PrintWindow(hwnd, memdc, 0)

    bmi = BITMAPINFOHEADER()
    bmi.biSize = ctypes.sizeof(BITMAPINFOHEADER)
    bmi.biWidth = w
    bmi.biHeight = -h            # 负数 = 自上而下
    bmi.biPlanes = 1
    bmi.biBitCount = 32
    bmi.biCompression = 0
    buf = ctypes.create_string_buffer(w * h * 4)
    got = g.GetDIBits(memdc, bmp, 0, h, buf, ctypes.byref(bmi), 0)

    nrows = h if rows is None else min(rows, h)
    rowsz = ((w * 3 + 3) // 4) * 4
    data = bytearray()
    for y in range(nrows - 1, -1, -1):          # BMP 自下而上存储
        src = y * w * 4
        row = bytearray(buf.raw[src:src + w * 4])
        del row[3::4]                           # 去掉 alpha 通道
        row += b"\x00" * (rowsz - w * 3)
        data += row

    hdr = struct.pack("<2sIHHI", b"BM", 14 + 40 + len(data), 0, 0, 54)
    info = struct.pack("<IiiHHIIiiII", 40, w, nrows, 1, 24, 0, len(data), 2835, 2835, 0, 0)
    with open(path, "wb") as f:
        f.write(hdr)
        f.write(info)
        f.write(data)

    g.DeleteObject(bmp)
    g.DeleteDC(memdc)
    u.ReleaseDC(hwnd, hdc)
    return w, nrows, bool(ok), got
