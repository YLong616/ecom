# ── 奖级判定（原有）──────────────────────────────────
def _prize(hit_front, hit_back):
    if hit_front == 5 and hit_back == 2: return "一等奖"
    if hit_front == 5 and hit_back == 1: return "二等奖"
    if hit_front == 5 and hit_back == 0: return "三等奖"
    if hit_front == 4 and hit_back == 2: return "四等奖"
    if hit_front == 4 and hit_back == 1: return "五等奖"
    if hit_front == 3 and hit_back == 2: return "六等奖"
    if hit_front == 4 and hit_back == 0: return "七等奖"
    if (hit_front == 3 and hit_back == 1) or (hit_front == 2 and hit_back == 2): return "八等奖"
    if (hit_front == 3 and hit_back == 0) or (hit_front == 1 and hit_back == 2) or (hit_front == 0 and hit_back == 2): return "九等奖"
    return "未中奖"

# ── 数据加载/保存（原有）────────────────────────────
def _load_history():
    if HIST_FILE.exists():
        return json.loads(HIST_FILE.read_text(encoding="utf-8"))
    return []

def _save_history(data):
    DATA_DIR.mkdir(parents=True, exist_ok=True)
    HIST_FILE.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")

# ── 1. 数据抓取（✅ 改为分页循环，获取全部）────────────
def fetch_from_sporttery_all(timeout=20):
    """
    从体彩官网 API 分页抓取全部历史数据（含奖金信息）
    返回所有记录列表
    """
    url = "https://webapi.sporttery.cn/gateway/lottery/getHistoryPageListV1.qry"
    all_records = []
    page_no = 1
    page_size = 100

    while True:
        params = {
            "gameNo": "85",
            "provinceId": "0",
            "pageSize": str(page_size),
            "isVerify": "1",
            "pageNo": str(page_no),
        }
        try:
            resp = requests.get(url, params=params, headers=HEADERS,
                               timeout=timeout, proxies=NO_PROXY)
            data = resp.json()
            if data.get("errorCode") != "0":
                raise Exception(data.get("errorMsg", "API 返回异常"))

            items = data.get("value", {}).get("list", [])
            if not items:
                break   # 无数据，结束循环

            for item in items:
                period = item.get("lotteryDrawNum", "")
                result_str = item.get("lotteryDrawResult", "")
                nums = [int(x) for x in result_str.split() if x.isdigit()]
                if len(nums) != 7:
                    continue

                # 奖级信息
                prize_list = item.get("prizeLevelList", [])
                first_count = first_area = second_count = third_count = ""
                for p in prize_list:
                    level = p.get("prizeLevel", "")
                    if "一等奖" in level:
                        first_count = p.get("winningNumber", "")
                        area = p.get("winningArea", "")
                        first_area = ",".join(area) if isinstance(area, list) else str(area)
                    elif "二等奖" in level:
                        second_count = p.get("winningNumber", "")
                    elif "三等奖" in level:
                        third_count = p.get("winningNumber", "")

                all_records.append({
                    "period": period,
                    "draw_time": item.get("lotteryDrawTime", ""),
                    "front": sorted(nums[:5]),
                    "back": sorted(nums[5:]),
                    "sales": item.get("sales", ""),
                    "prize_pool": item.get("prizePool", ""),
                    "first_win_count": first_count,
                    "first_area": first_area,
                    "second_win_count": second_count,
                    "third_win_count": third_count,
                })

            print(f"  已抓取第 {page_no} 页，共 {len(items)} 条，累计 {len(all_records)} 条")

            # 如果当前页不足 page_size，说明是最后一页
            if len(items) < page_size:
                break
            page_no += 1
            # 适当延时，避免请求过快
            import time
            time.sleep(0.5)

        except Exception as e:
            print(f"  第 {page_no} 页抓取失败: {e}")
            break

    return all_records


def cmd_fetch(_):
    DATA_DIR.mkdir(parents=True, exist_ok=True)
    print("=" * 58)
    print("  [Step 1] 联网抓取（全量）")
    print("=" * 58)

    records = fetch_from_sporttery_all(100)
    if not records:
        print("FETCH_FAILED:网络请求失败或 API 返回异常")
        return
    print(f"  sporttery.cn: 共获取 {len(records)} 期")

    # --- 精简版（分析用） ---
    existing = _load_history()
    exist_set = {r["period"] for r in existing}
    new_n = 0
    for r in records:
        simple = {"period": r["period"], "front": r["front"], "back": r["back"]}
        if r["period"] not in exist_set:
            existing.append(simple)
            new_n += 1
    existing.sort(key=lambda x: x["period"], reverse=True)
    _save_history(existing)

    # --- 完整版（导出用） ---
    raw_existing = _load_raw()
    raw_set = {r["period"] for r in raw_existing}
    new_raw = 0
    for r in records:
        if r["period"] not in raw_set:
            raw_existing.append(r)
            new_raw += 1
    raw_existing.sort(key=lambda x: x["period"], reverse=True)
    _save_raw(raw_existing)
    print(f"原始完整数据新增 {new_raw} 期，共 {len(raw_existing)} 期")

    if existing:
        lat = existing[0]
        fs = " ".join(f"{n:02d}" for n in lat["front"])
        bs = " ".join(f"{n:02d}" for n in lat["back"])
        print(f"最新: {lat['period']}  前区 {fs}  后区 {bs}")
        print(f"FETCH_OK:{lat['period']} (总期数 {len(existing)})")
    else:
        print("FETCH_FAILED: 无有效记录")

    # 自动复盘（如果新增数据）
    if new_n > 0 and len(existing) > 1:
        _auto_review(lat)


def _auto_review(latest_draw):
    """自动复盘（原有逻辑不变）"""
    arch = _load_archive()
    if not arch:
        return
    period = latest_draw["period"]
    act_front = latest_draw["front"]
    act_back = latest_draw["back"]

    target = None
    for entry in reversed(arch):
        if entry.get("review") is None and entry.get("actual") is None:
            target = entry
            break
    if not target:
        return

    print(f"\n── 自动复盘 {period} 期 ──────────────────────────────────")
    fs = " ".join(f"{n:02d}" for n in act_front)
    bs = " ".join(f"{n:02d}" for n in act_back)
    print(f"开奖: 前区 {fs}  后区 {bs}")

    results = []
    for p in target["predictions"]:
        hf = len(set(p["front"]) & set(act_front))
        hb = len(set(p["back"]) & set(act_back))
        pz = _prize(hf, hb)
        fs2 = " ".join(f"{n:02d}" for n in sorted(p["front"]))
        bs2 = " ".join(f"{n:02d}" for n in sorted(p["back"]))
        print(f"  #{p['idx']} 前{fs2} 后{bs2} → 前{hf}/5 后{hb}/2 {pz}")
        results.append({"idx": p["idx"], "hit_front": hf, "hit_back": hb, "prize": pz})

    target["actual"] = {"period": period, "front": act_front, "back": act_back}
    target["review"] = results
    _save_archive(arch)

# ── 导出功能（✅ 修改为保存到当前目录）────────────────
def cmd_export(args):
    """导出历史数据为 CSV 和 TXT（保存在当前目录）"""
    raw = _load_raw()
    if not raw:
        print("无完整原始数据，请先运行: python dlt.py fetch")
        sys.exit(1)

    raw_sorted = sorted(raw, key=lambda x: x["period"])

    rows = []
    for r in raw_sorted:
        front_str = " ".join(f"{n:02d}" for n in r["front"])
        back_str = " ".join(f"{n:02d}" for n in r["back"])
        draw_date = r.get("draw_time", "").split()[0] if r.get("draw_time") else ""
        rows.append({
            "期号": r["period"],
            "开奖日期": draw_date,
            "前区号码": front_str,
            "后区号码": back_str,
            "销售额": r.get("sales", ""),
            "奖池奖金": r.get("prize_pool", ""),
            "一等奖注数": r.get("first_win_count", ""),
            "一等奖地区": r.get("first_area", ""),
            "二等奖注数": r.get("second_win_count", ""),
            "三等奖注数": r.get("third_win_count", ""),
        })

    if not rows:
        print("没有有效记录")
        return

    ts = datetime.now().strftime("%Y%m%d_%H%M%S")
    base = f"dlt_history_{ts}"

    # ✅ 关键改动：使用当前工作目录（脚本所在目录）
    cwd = Path.cwd()
    csv_path = cwd / f"{base}.csv"
    txt_path = cwd / f"{base}.txt"

    with open(csv_path, "w", newline="", encoding="utf-8-sig") as f:
        writer = csv.DictWriter(f, fieldnames=EXPORT_FIELDS)
        writer.writeheader()
        writer.writerows(rows)

    with open(txt_path, "w", encoding="utf-8-sig") as f:
        f.write(",".join(EXPORT_FIELDS) + "\n")
        for row in rows:
            line = ",".join(str(row.get(k, "")) for k in EXPORT_FIELDS)
            f.write(line + "\n")

    print(f"✅ 导出完成：")
    print(f"   CSV: {csv_path}")
    print(f"   TXT: {txt_path}")
    print(f"   共 {len(rows)} 条记录")
